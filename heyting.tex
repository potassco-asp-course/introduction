\begin{frame}{Logic \`a la Arend Heyting}{and Kurt G\"odel}
  \smallskip
  \begin{itemize}
  \item \structure{Alphabet} \ A set $\mathcal{A}$ of atoms
  \item \structure{Formula} \
    \(
    \varphi \ ::= \ \bot \ \mid \ a \ \mid \ \varphi \land \varphi \ \mid \  \varphi \lor \varphi \ \mid \  \varphi \rightarrow \varphi
    \)
    \smallskip
  \item<2-> \structure{Interpretation} \ A pair \tuple{H,T} of sets of atoms with $H \subseteq T \subseteq\mathcal{A}$
    \begin{itemize}
    \item $H$ is called ``here'' and
    \item $T$ is called ``there''
    \end{itemize}
  \item<2->[] \structure{Note} \ \tuple{H,T} is a simplified Kripke structure
    \smallskip
  \item<3-> \structure{Intuition} \
    \begin{itemize}
    \item $H$ represents provably true atoms
    \item $T$ represents possibly true atoms
    \item $\mathcal{A}\setminus T$ represents false atoms
    \end{itemize}
  \end{itemize}
\end{frame}
% ----------------------------------------------------------------------
\begin{frame}{Logic \`a la Arend Heyting}{and Kurt G\"odel}
  \smallskip
  \begin{itemize}
  \item \structure{Satisfaction}
    \begin{itemize}
    \item $\tuple{H,T} \models a$ if $a \in H$
    \item $\tuple{H,T} \models \varphi \wedge \psi$ if
      $\tuple{H,T} \models \varphi$
      and
      $\tuple{H,T} \models \psi$
    \item $\tuple{H,T} \models \varphi \vee \psi$ if
      $\tuple{H,T} \models \varphi$
      or
      $\tuple{H,T} \models \psi$
      \smallskip
    \item<2-> \alert<2-3>{$\tuple{H,T} \models \varphi \rightarrow \psi$ if
        $\tuple{X,T} \not\models \varphi$ or $\tuple{X,T} \models \psi$
        \makebox[0pt][l]{for both $X=H,T$}}
      \smallskip
    \item<only@3> $\tuple{H,T} \models \neg\varphi$ if $\tuple{T,T} \not\models \varphi$
      \hfill ($\neg\varphi = \varphi\to\bot$)
    \end{itemize}
  \item<4-> \structure{Intuition}
    \begin{itemize}
    \item $\tuple{H,T}\models\varphi\ \sim\ \varphi$ is provably true
    \item $\tuple{T,T}\models\varphi\ \alt<5->{\text{\,iff\,}}{\sim}\ \only<5->{T\models}\varphi$\only<-4>{ is possibly true}
    \end{itemize}
    \smallskip
  \item<6-> \structure{Model} \ An interpretation \tuple{H,T} such that $\tuple{H,T} \models \varphi$
  \end{itemize}
  \medskip
  \only<7>{This system is called the \alert{Logic of Here-and-There} (\HT;~Heyting, 1930),\par also known as~G3~(G\"odel, 1932)}
\end{frame}
