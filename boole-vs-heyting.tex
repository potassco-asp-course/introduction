\begin{frame}{Boolean vs Heytingian Logic}
  \begin{itemize}
  \item \structure{Implication}
    \begin{itemize}
    \item \structure{Boolean} \
      $\varphi\rightarrow\phi$ is merely a logical shortcut for $\neg\varphi\vee\phi$
    \item \structure{Heytingian} \
      Implication represents a directional derivation
    \end{itemize}
    \smallskip
  \item \structure{Foundedness} \ Truth must be earned
    \begin{itemize}
    \item The truth of a formula is only justified if there exists a derivation starting from the facts
    \item This is fundamental property separating Boolean from Heytingian-style systems
    \end{itemize}
    \smallskip
  \item<2-> \structure{Tautologies}
    \begin{center}
    \begin{tabular}{c|c|c}
      & \structure{Boolean} & \structure{Heytingian}\\
      \hline
      $\varphi\vee\neg\varphi$        &$\checkmark$&$\times$\\
      $\neg\varphi\vee\neg\neg\varphi$&$\checkmark$&$\checkmark$\\
      $\neg\neg\varphi\to\varphi$     &$\checkmark$&$\times$\\
      $\varphi\to\neg\neg\varphi$     &$\checkmark$&$\checkmark$
    \end{tabular}
    \end{center}
  \end{itemize}
\end{frame}
