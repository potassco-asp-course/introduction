\begin{frame}{Logic \`a la George Boole}
  \smallskip
  \begin{itemize}
  \item \structure{Alphabet} \ A set $\mathcal{A}$ of atoms
  \item \structure{Formula}  \
    \(
    \varphi \ ::= \ \bot \ \mid \ a\in\mathcal{A} \ \mid \ \varphi \land \varphi \ \mid \  \varphi \lor \varphi \ \mid \  \varphi \rightarrow \varphi
    \)
  \item[] <2,4> \structure{Definition} \ $\neg\varphi = \varphi\to\bot$
  \item<3-> \structure{\alt<6>{\alert{Assignment}}{Interpretation}} \ A set $I$ of atoms with $I \subseteq \mathcal{A}$
  \item<3-> \structure{Intuition}      \ $I$ represents true atoms,
    $\mathcal{A}\setminus I$ represents false atoms
  \item<3-> \structure{Satisfaction}
  \begin{itemize}
  \item $I \models a$ if $a \in I$
  \item $I \models \varphi \wedge \psi$ if
    $I \models \varphi$
    and
    $I \models \psi$
  \item $I \models \varphi \vee \psi$ if
    $I \models \varphi$
    or
    $I \models \psi$
  \item $I \models \varphi \rightarrow \psi$ if
    $I \models \varphi$ implies $I \models \psi$
    \smallskip
  \item<4> $I \models \neg\varphi$ if $I \not\models \varphi$
  \end{itemize}
  \item<5-> \structure{\alt<6>{\alert{Solution}}{Model}} \ An \alt<6>{\alert{assignment}}{interpretation} $I$ such that $I \models \varphi$
  \end{itemize}
\end{frame}
